.build1:
  stage: build1
  <<: *manual-rules
  needs:
    - job: prepare:tag1
      artifacts: true
  tags:
    - docker
    - $RUNNER_TAG 
  script:
  - |
    echo "Starting build script..."
    PHASE="${PHASE_NAME:-testphase1}"               
    USER_TAG="${IMAGE_TAG:-ltst}"   
    REPO_URL="https://gitlab-ci-token:${CI_JOB_TOKEN}@${CI_SERVER_HOST}/${CI_PROJECT_NAMESPACE}/${REPO_SLUG}.git"
    SERVICE_BRANCH="${SERVICE_BRANCH:-main}"
    echo "Cloning ${CI_PROJECT_NAMESPACE}/${REPO_SLUG} @ ${SERVICE_BRANCH} ..."
    git clone \
      --depth 1 \
      --branch "${SERVICE_BRANCH}" \
      "$REPO_URL" \
      repo
    echo "HEAD: $(git -C repo rev-parse --short HEAD)"

    

    # ---------------------------
    # Build coordinates
    # --------------------------
    IMAGE_BASE="${CI_REGISTRY}/${CI_PROJECT_PATH}/${PHASE}/${IMAGE_NAME}"
    IMAGE_TAG=$IMAGE_BASE
    SERVICE_SHA=$(git -C repo rev-parse HEAD)
    SERVICE_URL="https://${CI_SERVER_HOST}/${CI_PROJECT_NAMESPACE}/${REPO_SLUG}"

    # 3. Pull cache image — :latest on main, nothing on other branches
    #    (other branches only push :<branch>-<sha>, no floating tag to pull)
    
    #if [ "$CI_COMMIT_REF_NAME" = "main" ]; then
     # docker pull "${IMAGE_BASE}:latest" 2>/dev/null || true
    #fi

    # 4. Build with BuildKit
    #    Every service repo has a Dockerfile at its root.
    #    Build context = repo/  ->  Docker finds repo/Dockerfile automatically.
    echo "Starting Docker Build..."
    #CACHE_FROM="${IMAGE_BASE}:latest"

    export DOCKER_BUILDKIT=1
    BUILD_SECRET_ARG=""

    if [ "$USE_BUILD_SECRET" = "true" ]; then
      BUILD_SECRET_ARG="--secret id=ci_job_token,env=CI_JOB_TOKEN"
    fi

    docker build \
      $BUILD_SECRET_ARG \
      --platform "${BUILD_PLATFORM}" \
      --build-arg BUILDKIT_INLINE_CACHE=1 \
      --build-arg GIT_COMMIT="${CI_COMMIT_SHORT_SHA}" \
      --build-arg GIT_BRANCH="${CI_COMMIT_REF_NAME}" \
      --build-arg BUILD_DATE="${TIMESTAMP}" \
      -t "${IMAGE_BASE}:${USER_TAG}-${TIMESTAMP}" \
      repo

    

      #--cache-from "${IMAGE_BASE}:latest" \
      #--build-arg  BUILD_DATE="$(date -u +%Y-%m-%dT%H:%M:%SZ)"
        
    # ------------------------------------------------------
    # Tag timestamp version for non-main branch
    # ------------------------------------------------------
    #docker tag "${IMAGE_BASE}:${USER_TAG}" "${IMAGE_BASE}:${USER_TAG}-${TIMESTAMP}"
    
    echo "Image ready to push : ${IMAGE_BASE}:${USER_TAG}-${TIMESTAMP}"

    docker push "${IMAGE_BASE}:${USER_TAG}-${TIMESTAMP}"
    echo "Build Completed: ${IMAGE_BASE}:${USER_TAG}-${TIMESTAMP}"


    # 5. Tag :latest on main only
   # if [ "$CI_COMMIT_REF_NAME" = "main" ]; then
    #  docker tag "${IMAGE_BASE}:${USER_TAG}" "${IMAGE_BASE}:latest"
    #fi

    # 6. Push
    #    main branch  ->  :main-<sha>  :latest
    #    other branch ->  :branchname-<sha>
    #docker push "${IMAGE_BASE}:${USER_TAG}"
    
    
    #if [ "$CI_COMMIT_REF_NAME" = "main" ]; then
     # docker push "${IMAGE_BASE}:latest"
    #fi

    #echo "Build Completed: ${IMAGE_BASE}:${USER_TAG}"
    

  after_script:
    - rm -rf repo   # free runner disk space

          echo "Failed to fetch primary pod from cluster status"; exit 1

.sonar:
  stage: sonar
  <<: *schedule-rules
  needs:
    - job: prepare:tag
      artifacts: true
  before_script: []   # skip docker login
  tags:
    - shell-runner
  script:
  - |
 
    REPO_URL="https://gitlab-ci-token:${CI_JOB_TOKEN}@${CI_SERVER_HOST}/${CI_PROJECT_NAMESPACE}/${REPO_SLUG}.git"
    SSANITY_SERVICE_BRANCH="${SSANITY_SERVICE_BRANCH}"

    git clone \
      --depth 1 \
      --branch "${SANITY_SERVICE_BRANCH}" \
      "$REPO_URL" \
      repo
    echo "HEAD: $(git -C repo rev-parse --short HEAD)"

    cd repo

    # Dynamic extraction
    SONAR_NAME="${REPO_SLUG}"
    SONAR_KEY="${REPO_SLUG##*-}"



    if [ -f "pom.xml" ]; then
        echo " Java project detected"

        # Using HOST Maven
        export MAVEN_HOME="/usr/share/maven"
        export JAVA_HOME="/usr/lib/jvm/java-21-amazon-corretto.x86_64"
        export PATH="$JAVA_HOME/bin:$MAVEN_HOME/bin:$PATH"

        mvn clean verify sonar:sonar -DskipTests \
          -Dsonar.projectKey="${SONAR_KEY}" \
          -Dsonar.projectName="${SONAR_NAME}" \
          -Dsonar.host.url="${SONAR_HOST_URL}" \
          -Dsonar.login="${SONAR_TOKEN}"

      else
        echo " Non-Java project detected"

        sonar-scanner \
          -Dsonar.projectKey="${SONAR_KEY}" \
          -Dsonar.projectName="${SONAR_NAME}" \
          -Dsonar.exclusions=**/*.java \
          -Dsonar.host.url="${SONAR_HOST_URL}" \
          -Dsonar.login="${SONAR_TOKEN}" \
          ${SONAR_EXTRA_ARGS}
      fi

      echo ">>> SonarQube report generation"

      PS=500
      PAGE=1
      COLLECTED=0

      mkdir -p reports
      # initialize json file
      : > reports/issues.json

      PS=500
      PAGE=1
      COLLECTED=0

      while true; do
        RESPONSE=$(curl -s -H "Authorization: Bearer $SONAR_TOKEN" \
          "${SONAR_HOST_URL%/}/api/issues/search?componentKeys=$SONAR_KEY&resolved=false&types=BUG,CODE_SMELL,VULNERABILITY&ps=$PS&p=$PAGE")

        TOTAL=$(echo "$RESPONSE" | jq '.paging.total')
        COUNT=$(echo "$RESPONSE" | jq '.issues | length')

        echo "$RESPONSE" | jq -c '.issues[]' >> reports/issues.json

        COLLECTED=$((COLLECTED + COUNT))
        echo "Fetched $COLLECTED / $TOTAL issues"

        [ "$COLLECTED" -ge "$TOTAL" ] && break
        PAGE=$((PAGE + 1))
      done


      jq -n -r '
        (["key","rule","type","severity","component","line","message"] | @csv),
        (inputs | [
          .key,
          .rule,
          .type,
          .severity,
          .component,
          (.line // ""),
          (.message // "" | gsub("\n"; " "))
        ] | @csv)
      ' reports/issues.json >> reports/issues.csv

      ISSUES_JSON="reports/issues.json"

      BLOCKER=$(jq -c 'select(.severity=="BLOCKER")'  "$ISSUES_JSON" | wc -l)
      CRITICAL=$(jq -c 'select(.severity=="CRITICAL")' "$ISSUES_JSON" | wc -l)
      MAJOR=$(jq -c 'select(.severity=="MAJOR")'    "$ISSUES_JSON" | wc -l)
      MINOR=$(jq -c 'select(.severity=="MINOR")'    "$ISSUES_JSON" | wc -l)
      INFO=$(jq -c 'select(.severity=="INFO")'     "$ISSUES_JSON" | wc -l)

      TOTAL=$((BLOCKER + CRITICAL + MAJOR + MINOR + INFO))

      
      echo "-----------------------------------------"
      echo " SonarQube Overall Issues (GUI Match)"
      echo "-----------------------------------------"
      echo " TOTAL:     $TOTAL"
      echo " BLOCKER:  $BLOCKER"
      echo " CRITICAL: $CRITICAL"
      echo " MAJOR:    $MAJOR"
      echo " MINOR:    $MINOR"
      echo " INFO:     $INFO"
      echo "-----------------------------------------"


      SUMMARY_FILE="reports/summary-$REPO_SLUG.txt"
      
      echo "-----------------------------------------" >> "$SUMMARY_FILE"
      echo " SERVICE: $REPO_SLUG" >> "$SUMMARY_FILE"
      echo "-----------------------------------------" >> "$SUMMARY_FILE"
      echo " TOTAL:     $TOTAL" >> "$SUMMARY_FILE"
      echo " BLOCKER:   $BLOCKER" >> "$SUMMARY_FILE"
      echo " CRITICAL:  $CRITICAL" >> "$SUMMARY_FILE"
      echo " MAJOR:     $MAJOR" >> "$SUMMARY_FILE"
      echo " MINOR:     $MINOR" >> "$SUMMARY_FILE"
      echo " INFO:      $INFO" >> "$SUMMARY_FILE"
      echo "" >> "$SUMMARY_FILE"



  artifacts:
    when: always
    name: "SonarQube-Report"
    paths:
      - repo/reports/
    expire_in: 10 days

        TOTAL_TESTS=$(grep -h '<testsuite ' target/surefire-reports/*.xml \
          | sed -n 's/.*tests="\([0-9]*\)".*/\1/p' \
          | awk '{s+=$1} END {print s}')

        TOTAL_FAILURES=$(grep -h '<testsuite ' target/surefire-reports/*.xml \
          | sed -n 's/.*failures="\([0-9]*\)".*/\1/p' \
          | awk '{s+=$1} END {print s}')

        TOTAL_ERRORS=$(grep -h '<testsuite ' target/surefire-reports/*.xml \
          | sed -n 's/.*errors="\([0-9]*\)".*/\1/p' \
          | awk '{s+=$1} END {print s}')

        TOTAL_SKIPPED=$(grep -h '<testsuite ' target/surefire-reports/*.xml \
         | sed -n 's/.*skipped="\([0-9]*\)".*/\1/p' \
         | awk '{s+=$1} END {print s}')

      fi

        echo " Code coverage meets threshold(≥ 85%) "
      fi

      # Step 6:  Final pipeline status
      # -------------------------------
      if [ "$PIPELINE_FAILED" -eq 1 ]; then
        echo "PIPELINE FAILED"
        exit 1
      else
        echo "PIPELINE PASSED"
      fi

# ==========================================================
# REPORT + QUALITY GATE (JSON → CSV + Threshold)
# ==========================================================
.sonarqube_report:
  stage: report
  image: alpine:3.20
  tags:
    - docker
    - $RUNNER_TAG
  before_script:
    - apk add --no-cache curl jq
 
  rules:
    - if: '$CI_PIPELINE_SOURCE == "merge_request_event" && $CI_MERGE_REQUEST_TARGET_BRANCH_NAME == "main"'
      when: on_success
    - when: never
  needs:
    - job: scan:nem
      optional: true
      
    - job: scan:nem:nonjava
      optional: true
 
  script:
    - |
        set -euo pipefail
        echo ">>> SonarQube report (pagination + NDJSON)"
        PS=500
        PAGE=1
        COLLECTED=0
        mkdir -p reports
        : > reports/issues.json

        while true; do
          RESPONSE=$(curl -s -H "Authorization: Bearer $SONAR_ENT_TOKEN" \
            "${SONAR_ENT_HOST_URL%/}/api/issues/search?componentKeys=$SONAR_KEY&branch=${CI_COMMIT_REF_NAME}&resolved=false&types=BUG,CODE_SMELL,VULNERABILITY&ps=$PS&p=$PAGE")

          TOTAL=$(echo "$RESPONSE" | jq '.paging.total')
          COUNT=$(echo "$RESPONSE" | jq '.issues | length')

          echo "$RESPONSE" | jq -c '.issues[]' >> reports/issues.json

          COLLECTED=$((COLLECTED + COUNT))
          echo "Fetched $COLLECTED / $TOTAL issues"

          [ "$COLLECTED" -ge "$TOTAL" ] && break
          PAGE=$((PAGE + 1))
        done

        # NDJSON -> CSV (header once)
        jq -n -r '
          (["key","rule","type","severity","component","line","message"] | @csv),
          (inputs | [
            .key,
            .rule,
            .type,
            .severity,
            .component,
            (.line // ""),
            (.message // "" | gsub("\n"; " "))
          ] | @csv)
        ' reports/issues.json > reports/issues.csv

        ISSUES_JSON="reports/issues.json"

        BLOCKER=$(jq -c 'select(.severity=="BLOCKER")'  "$ISSUES_JSON" | wc -l)
        CRITICAL=$(jq -c 'select(.severity=="CRITICAL")' "$ISSUES_JSON" | wc -l)
        MAJOR=$(jq -c 'select(.severity=="MAJOR")'    "$ISSUES_JSON" | wc -l)
        MINOR=$(jq -c 'select(.severity=="MINOR")'    "$ISSUES_JSON" | wc -l)
        INFO=$(jq -c 'select(.severity=="INFO")'     "$ISSUES_JSON" | wc -l)

        TOTAL=$((BLOCKER + CRITICAL + MAJOR + MINOR + INFO))

        echo "-----------------------------------------"
        echo " SonarQube Overall Issues (GUI Match)"
        echo "-----------------------------------------"
        echo " TOTAL:     $TOTAL"
        echo " BLOCKER:  $BLOCKER"
        echo " CRITICAL: $CRITICAL"
        echo " MAJOR:    $MAJOR"
        echo " MINOR:    $MINOR"
        echo " INFO:     $INFO"
        echo "-----------------------------------------"

        # -------------------------------
        # 4) QUALITY GATE
        # -------------------------------
        
        MAX_MINOR=20

        if [ "$BLOCKER" -ne 0 ] || \
             [ "$CRITICAL" -ne 0 ] || \
             [ "$MAJOR" -ne 0 ] || \
             [ "$MINOR" -gt "$MAX_MINOR" ]; then


            echo "Merge blocked due to quality gate failure:"

            echo "   Merge blocked due to quality gate failure:"
            echo "   BLOCKER=$BLOCKER (must be 0)"
            echo "   CRITICAL=$CRITICAL (must be 0)"
            echo "   MAJOR=$MAJOR (must be 0)"
            echo "   MINOR=$MINOR (must be ≤ $MAX_MINOR)"
            exit 1
        else
          echo "Quality gate passed. Please proceed with Merge Request"
        fi

  artifacts:
    when: always
    name: "SonarQube-Report"
    paths:
      - reports/
    expire_in: 10 days

deploy:
  stage: deploy
  needs:
    - job: build_and_push_image
      artifacts: true

  tags:
    - shell-runner

  rules:
    - if: '$CI_PIPELINE_SOURCE == "push" && $CI_COMMIT_BRANCH == "main"'

    - when: never


  script:
    - |
      echo "Starting deployment..."

      APP_NAME="$CI_PROJECT_NAME"

      # Validate required variables
      if [ -z "$APP_NAME" ] || [ -z "$IMAGE_NAME" ] || [ -z "$IMAGE_TAG" ]; then
        echo "ERROR: Required variables not set (APP_NAME, IMAGE_NAME, IMAGE_TAG)"
        exit 1
      fi

      # Move to manifest directory
      cd ../opnms3ps-platform/manual-deployment/opnms3ps_manifest_templates || exit 1

      echo "Looking for manifest file for app: $APP_NAME"

      # Find matching manifest file
      MANIFEST_FILE="${APP_NAME}-manifest.yaml"

      if [ -z "$MANIFEST_FILE" ]; then
        echo "ERROR: No manifest file found for $APP_NAME"
        exit 1
      fi

      echo "Found manifest: $MANIFEST_FILE"

      # Construct full image
      FULL_IMAGE="$IMAGE_NAME:$IMAGE_TAG"
      echo "Updating image to: $FULL_IMAGE"

      NAMESPACE_NAME="ci-ns"
      # Updating manifest image and namespace
      sed -i \
        -e "s|image: .*${APP_NAME}:.*|image: ${FULL_IMAGE}|g" \
        -e "s|namespace:.*|namespace: ${NAMESPACE_NAME}|g" \
        "$MANIFEST_FILE"

      echo "Updated manifest preview:"
      grep "namespace:" "$MANIFEST_FILE"
      grep "image:" "$MANIFEST_FILE"

      # Deploy
      echo "Applying manifest..."
      kubectl apply -f "$MANIFEST_FILE" -n $NAMESPACE_NAME
      
      sleep 10

      echo "Waiting for rollout to complete..."
      kubectl rollout status deployment/$APP_NAME -n "$NAMESPACE_NAME" --timeout=300s
      output=$?

      echo "Deployment exit code: $output"

      if [ $output -ne 0 ]; then
        echo " Deployment FAILED"
        exit 1
      else
        kubectl get pods -n "$NAMESPACE_NAME" -o custom-columns="NAMESPACE:.metadata.namespace,POD:.metadata.name,IMAGE:.spec.containers[*].image,STATUS:.status.phase" | grep "$APP_NAME"

        echo " Deployment successful and stable"
      fi

.mr-sanity-build:
  stage: mr-sanity-build
  <<: *trigger-rules
  needs:
    - job: mr-sanity-prepare:tag
      artifacts: true   
  tags:
    - docker
    - $RUNNER_TAG
  script:
  - |

    echo "Platform PIPELINE triggered"


    REPO_URL="https://gitlab-ci-token:${CI_JOB_TOKEN}@${CI_SERVER_HOST}/${CI_PROJECT_NAMESPACE}/${REPO_SLUG}.git"


    if [ "${REPO_SLUG}" = "${FULL_APP_NAME}" ]; then
      MR_SERVICE_BRANCH="${BRANCH_NAME}"
      MR_IMAGE_TAG="${PROJECT_NAME}-test-${MR_IID}-${TIMESTAMP}"
    else
      MR_SERVICE_BRANCH="main"
      MR_IMAGE_TAG="${PROJECT_NAME}-main-${MR_IID}-${TIMESTAMP}"
    fi

      echo "Cloning ${CI_PROJECT_NAMESPACE}/${REPO_SLUG} @ ${MR_SERVICE_BRANCH} ..."
      git clone \
        --depth 1 \
        --branch "${MR_SERVICE_BRANCH}" \
        "$REPO_URL" \
        repo
    
     
      echo "Starting Docker Sanity-build..."
  
      IMAGE_NAME="${CI_REGISTRY}/${CI_PROJECT_PATH}/${MR_PHASE_NAME}/${REPO_SLUG}:${MR_IMAGE_TAG}"

      echo "Sanity building image ${IMAGE_NAME}"

      export DOCKER_BUILDKIT=1
      BUILD_SECRET_ARG=""

      if [ "$USE_BUILD_SECRET" = "true" ]; then
        BUILD_SECRET_ARG="--secret id=ci_job_token,env=CI_JOB_TOKEN"
      fi

      docker build \
        $BUILD_SECRET_ARG \
        --platform "${BUILD_PLATFORM}" \
        --build-arg BUILDKIT_INLINE_CACHE=1 \
        --build-arg GIT_COMMIT="${CI_COMMIT_SHORT_SHA}" \
        --build-arg GIT_BRANCH="${MR_SERVICE_BRANCH}" \
        --build-arg BUILD_DATE="${TIMESTAMP}" \
        -t "${IMAGE_NAME}" \
        repo

      echo "MR_IMAGE_TAG=${MR_IMAGE_TAG}" >> "${CI_PROJECT_DIR}/sanity-build.env"

      
      docker push "${IMAGE_NAME}"

      echo "Sanity build Completed: ${IMAGE_NAME}"

     
  artifacts:
    reports:
      dotenv: "${CI_PROJECT_DIR}/sanity-build.env"    # exports RELEASE_TAG into all downstream jobs
    expire_in: 1 day

  after_script:
    - rm -rf repo   # free runner disk space

mr-sanity-deploy:helm:
  stage: mr-sanity-deploy
  <<: *trigger-rules
  resource_group: locking-pipeline    # ← lock held for ENTIRE child pipeline
  tags:
    - shell-runner
  variables:
    BUILD_PLATFORM: "linux/amd64"
    GIT_STRATEGY: clone
 
  needs:
    - job: mr-sanity-prepare:tag
      artifacts: true
    - mr-sanity-build:service-aus
    - mr-sanity-build:service-wui
    - mr-sanity-build:service-agw
    - mr-sanity-build:service-cfm
    - mr-sanity-build:service-ewi
    - mr-sanity-build:service-nem
    - mr-sanity-build:service-wbs
    - mr-sanity-build:service-nbi
    - mr-sanity-build:service-rpm
  


  before_script:
    - export KUBECONFIG=/home/gitlab-runner/.kube/config

  script:
    - |

      PLATFORM_URL="https://gitlab-ci-token:${CI_JOB_TOKEN}@${CI_SERVER_HOST}/${CI_PROJECT_NAMESPACE}/opnms3ps-platform.git"
      
      git clone "$PLATFORM_URL" repo_dir

      # Navigate to helm chart
      cd repo_dir/helm/opnms3ps-stack
      echo "Current directory: $(pwd)"
      ls -l      
    

      if helm list -n ${NAMESPACE_NAME} | grep -q opnms3ps-stack; then
        echo "Uninstalling opnms3ps-stack "
        helm uninstall opnms3ps-stack -n ${NAMESPACE_NAME}
        sleep 30
      fi

      

      BASE_PATH="${CI_REGISTRY}/${CI_PROJECT_PATH}"

       # update all imageBase
      sed -i "s|imageBase: \".*\"|imageBase: \"${BASE_PATH}/${MR_PHASE_NAME}\"|g" imageTag.yaml

      # Recreate tags exactly as build jobs created them

      TEST_TAG="${PROJECT_NAME}-test-${MR_IID}-${TIMESTAMP}"
      MAIN_TAG="${PROJECT_NAME}-main-${MR_IID}-${TIMESTAMP}"

      

      # Set all services to main tag
      sed -i "s|tag: \".*\"|tag: \"${MAIN_TAG}\"|g" imageTag.yaml

      # TEST_TAG will be updated only to application which triggeres the sanity pipeline
      
      sed -i "/^${FULL_APP_NAME}:/,/^[^[:space:]]/ s|tag: \".*\"|tag: \"${TEST_TAG}\"|" imageTag.yaml

      echo "Updated imageTag.yaml"

      PRIMARY_POD=$(kubectl get cluster opnms3ps-postgres -n opnms3ps-postgres-ns -o jsonpath='{.status.currentPrimary}')
 
      if [ -z "$PRIMARY_POD" ]; then
        echo "Failed to fetch primary pod from cluster status"; exit 1
      fi
  
      # DROP DATABASE - ${DATABASE_NAME}
      echo "DROPPING DATABASE ${DATABASE_NAME}"
      kubectl exec -n opnms3ps-postgres-ns ${PRIMARY_POD} -c postgres -- psql -U postgres -c "DROP DATABASE IF EXISTS \"$DATABASE_NAME\"  WITH (FORCE);"
      sleep 30s

      echo "CREATING DATABASE ${DATABASE_NAME}"
      kubectl exec -n opnms3ps-postgres-ns ${PRIMARY_POD} -c postgres -- psql -U postgres -c "CREATE DATABASE \"$DATABASE_NAME\" OWNER nmsuser;"
      sleep 30s


      
      helm dependency update
      sleep 10

      helm upgrade --install opnms3ps-stack . -f values.yaml -f imageTag.yaml \
        --namespace ${NAMESPACE_NAME} \
        --create-namespace \
        --set global.namespace="${NAMESPACE_NAME}"

      if [ $? -ne 0 ]; then
          echo "Helm deployment failed"
          exit 1
      fi


      echo "Waiting for all deployments to rollout..."

      kubectl get deploy -n ${NAMESPACE_NAME} -o name
      sleep 5

      kubectl rollout status deployment -n ${NAMESPACE_NAME} --timeout=300s
      output=$?

      echo "Deployment exit code: $output"

      if [ $output -ne 0 ]; then
        echo " Deployment not healthy"
        echo "Uninstalling Deployment"
        exit 1
      else
        kubectl get pods -o jsonpath="{range .items[*]}{.metadata.name}{': '}{.spec.containers[*].image}{'\n'}{end}" -n ${NAMESPACE_NAME} | grep -v kafka
        echo " Deployment successful and stable"
      fi

      echo "Waiting for 10 minutes..."
      for i in {9..0}; do
        sleep 60
        echo "$i minutes left"
      done

      RESTARTING=$(kubectl get pods -n ${NAMESPACE_NAME} --no-headers | grep -v "kafka" | awk '{sum += $4} END {print sum}')

      if [ "$RESTARTING" -gt 0 ]; then
        echo "Pods are restarting. Hence pipeline failed"
        exit 1
      fi

      # Wait for ingress to be available
      until kubectl get ingress -n ${NAMESPACE_NAME}; do
        echo "Waiting for ingress..."
        sleep 5
      done

      # Get ingress host dynamically
      INGRESS_HOST=$(kubectl get ingress -n "${NAMESPACE_NAME}" -o jsonpath='{.items[0].status.loadBalancer.ingress[0].hostname}')

      if [ -z "$INGRESS_HOST" ]; then
        echo "ERROR: Could not retrieve ingress hostname"
        exit 1
      fi

    
      APP_URL="https://${INGRESS_HOST}"

      echo "Sanity testing URL: $APP_URL"
      echo "APP_URL =$APP_URL"  >> "${CI_PROJECT_DIR}/sanity-build.env"
      
  artifacts:
    reports:
      dotenv: "${CI_PROJECT_DIR}/sanity-build.env"
